def painel
  @aba             = params[:aba].presence || "recargas"
  @recargas        = current_estudante.recargas.order(:data)
  @refeicoes       = current_estudante.refeicoes.order(:data)
  @total_recargas  = total_de(@recargas)
  @total_refeicoes = total_de(@refeicoes)
end
